# GPU-only dependencies for the Qwen3-TTS voice-clone worker subprocess
# and the faster-whisper transcription worker subprocess.
#
# These packages are ONLY needed by `voice_clone.runtime` and
# `media_processing.transcription_worker` (the worker subprocesses spawned
# by the services). The FastAPI app itself (`app.py`) never imports them,
# so a machine without a CUDA GPU can run the dashboard and the recording
# pipeline without this file.
#
# Verified on the target hardware (status REAL_MODEL_VERIFIED):
#
#   * GPU:  NVIDIA GeForce RTX 5070 (12 GB VRAM, Blackwell)
#   * OS:   Windows
#   * Python: 3.12.10
#
# IMPORTANT: the default PyPI wheel of `torch` is CPU-only on Windows.
# Installing `torch` from the regular index silently disables CUDA and
# the runtime will report `cuda_available=false`. The CUDA build MUST be
# pulled from the cu128 index. Use the exact command below, which pins
# the verified wheel:
#
#   python -m pip install -r requirements-gpu.txt \
#       --extra-index-url https://download.pytorch.org/whl/cu128
#
# Verified package versions:
#
#   qwen-tts      == 0.1.1
#   torch         == 2.11.0+cu128
#   transformers  == 4.57.3
#   accelerate    == 1.12.0
#
# faster-whisper reuses the same torch wheel for CUDA. It is imported
# lazily by the transcription worker only.
#
#   faster-whisper == 1.1.0
#
# `safetensors` is required by `transformers` to load the model weights.
# The spike did not pin a specific version; it was pulled in transitively
# by `transformers==4.57.3` and worked. We do NOT invent a pin here - the
# version resolved by `transformers==4.57.3` is the verified one.
#
# `flash-attn` is intentionally NOT pinned: on RTX 50-series (Blackwell)
# prebuilt wheels are unreliable. The runtime probes flash_attention_2
# first and falls back to sdpa automatically; a flash-attn failure never
# produces a fake result.

qwen-tts==0.1.1
torch==2.11.0+cu128
transformers==4.57.3
accelerate==1.12.0
# Let transformers pull a compatible safetensors wheel; do not force a
# version that has not been tested on the RTX 5070.
safetensors>=0.4.0

# faster-whisper for GPU transcription (imported lazily by the
# transcription worker subprocess only).
# Pinned to >=1.1.0 because the multilingual ASR presets forward the
# `multilingual`, `hallucination_silence_threshold` and `hotwords`
# parameters, which are only supported from 1.1.0 onward. The runtime
# compatibility check in media_processing.asr_presets refuses to forward
# any parameter the installed version does not understand.
faster-whisper==1.2.1

# nvidia-ml-py for backend-independent VRAM measurement via NVML.
# Used by the ASR benchmark worker to measure actual GPU memory usage
# regardless of whether the model uses CTranslate2 (Whisper) or PyTorch
# (NeMo). If NVML is not available, VRAM values are reported as null,
# never as 0. The package provides the `pynvml` module.
nvidia-ml-py>=11.5.0

# NeMo ASR toolkit for Parakeet TDT and Canary models.
# Imported lazily by the benchmark worker only — the FastAPI app starts
# without NeMo. Models are downloaded from HuggingFace on first use.
nemo_toolkit[asr]>=2.7.0

# diffusers for the video-generation worker subprocess (CogVideoX).
# Imported lazily by ttvturbo.video_generation.worker only — the FastAPI
# app starts without diffusers and reports `available=false` for video
# generation. Reuses the same torch+cu128 wheel pinned above; diffusers
# itself is CPU-agnostic and pulls no extra CUDA runtime.
diffusers>=0.30.0
# Pillow is required by diffusers for image loading (I2V source frames).
# Usually pulled in transitively by transformers; pinned here so the
# video-generation worker always has it.
Pillow>=10.0.0
